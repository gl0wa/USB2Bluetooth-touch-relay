cmake_minimum_required(VERSION 3.16)
project(touchcore_tests C)
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Werror -fsanitize=address,undefined)
add_link_options(-fsanitize=address,undefined)
set(CORE ../../components/touchcore)
add_executable(test_touchcore test_touchcore.c ${CORE}/hid_touch.c ${CORE}/touch_map.c)
target_include_directories(test_touchcore PRIVATE ${CORE}/include)
enable_testing()
add_test(NAME touchcore COMMAND test_touchcore)
